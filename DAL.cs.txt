using System;
using System.Linq;
using System.Text;
using System.Text.Json;
using System.Threading.Tasks;
using Capstone.API.DTOs.Summary;
using Capstone.API.Services.Interfaces;
using Capstone.DAL.DTOs;
using Capstone.DAL.Repository;
using Microsoft.Extensions.Configuration;

namespace Capstone.API.Services.Implementations;

public class SummaryService : ISummaryService
{
    private readonly IRepository _repository;
    private readonly IAuthenticationService _authService;
    private readonly IConfiguration? _configuration;

    public SummaryService(
        IRepository repository,
        IAuthenticationService authService,
        IConfiguration? configuration = null)
    {
        _repository = repository;
        _authService = authService;
        _configuration = configuration;
    }

    public async Task<SummaryResponse?> GetUserSummaryAsync(string actingUserName, string? targetUserName = null)
    {
        if (string.IsNullOrWhiteSpace(actingUserName) || !_authService.IsAdmin(actingUserName))
        {
            return null;
        }

        var effectiveUser = string.IsNullOrWhiteSpace(targetUserName) ? actingUserName : targetUserName;
        return await _repository.GetSummaryRequestByUserAsync(effectiveUser);
    }

    public async Task<AISummaryResponseDto?> GetAISummaryAsync(string actingUserName, string? targetUserName = null)
    {
        var data = await GetUserSummaryAsync(actingUserName, targetUserName);
        if (data is null)
        {
            return null;
        }

        var commonIssuesList = data.Requests
            .Where(r => !string.IsNullOrWhiteSpace(r.Description))
            .GroupBy(r => r.Description.Trim(), StringComparer.OrdinalIgnoreCase)
            .OrderByDescending(g => g.Count())
            .Take(3)
            .Select(g => $"{g.Key} ({g.Count()})")
            .ToList();

        var commonIssues = commonIssuesList.Count > 0 ? string.Join(", ", commonIssuesList) : "None";
        var scopeLabel = string.IsNullOrWhiteSpace(targetUserName)
            ? $"user '{data.UserName}'"
            : $"user '{targetUserName}'";

        var overallSummary =
            $"Summary for {scopeLabel}: {data.TotalRequests} total request(s), {data.OpenRequests} open, " +
            $"{data.InProgressRequests} in progress, {data.ClosedRequests} closed, and {data.ReOpenedRequests} reopened.";

        var recommendedActions = data.OpenRequests > 0 || data.ReOpenedRequests > 0
            ? "Prioritize open and reopened service requests and verify pending items."
            : "No open or reopened requests requiring immediate action.";

        var tableMarkdown = await TryGenerateFoundryTableAsync(data)
            ?? BuildSummaryTable(data, commonIssues, overallSummary, recommendedActions);

        return new AISummaryResponseDto
        {
            TotalRequests = data.TotalRequests,
            OpenRequests = data.OpenRequests,
            ClosedRequests = data.ClosedRequests,
            CommonIssues = commonIssues,
            OverallSummary = $"{overallSummary}{Environment.NewLine}{Environment.NewLine}{tableMarkdown}",
            RecommendedNextActions = recommendedActions
        };
    }

    public string BuildAgentPrompt(SummaryResponse data)
    {
        var sb = new StringBuilder();
        sb.AppendLine("Summarize the user's service requests in a Markdown table with these exact rows:");
        sb.AppendLine("1. Total Requests");
        sb.AppendLine("2. Open Requests");
        sb.AppendLine("3. Close Requests");
        sb.AppendLine("4. Common Issues");
        sb.AppendLine("5. Overall Summary");
        sb.AppendLine("6. Recommended Next Actions");
        sb.AppendLine();
        sb.AppendLine(JsonSerializer.Serialize(data));
        return sb.ToString();
    }

    protected virtual Task<string?> InvokeFoundryAgentAsync(string prompt, SummaryResponse data)
    {
        // Replace with Azure.AI.Projects AIProjectClient + DefaultAzureCredential call
        // once ProjectEndpoint and AgentId are populated in appsettings.json.
        return Task.FromResult<string?>(null);
    }

    private async Task<string?> TryGenerateFoundryTableAsync(SummaryResponse data)
    {
        var enabled = _configuration?.GetValue<bool>("FoundryAgent:Enabled") ?? false;
        var endpoint = _configuration?["FoundryAgent:ProjectEndpoint"];
        var agentId = _configuration?["FoundryAgent:AgentId"];

        if (!enabled || string.IsNullOrWhiteSpace(endpoint) || string.IsNullOrWhiteSpace(agentId))
        {
            return null;
        }

        try
        {
            var prompt = BuildAgentPrompt(data);
            var output = await InvokeFoundryAgentAsync(prompt, data);
            return string.IsNullOrWhiteSpace(output) ? null : output.Trim();
        }
        catch
        {
            return null;
        }
    }

    private static string BuildSummaryTable(
        SummaryResponse data,
        string commonIssues,
        string overallSummary,
        string recommendedActions)
    {
        var sb = new StringBuilder();
        sb.AppendLine("| # | Metric | Value |");
        sb.AppendLine("| --- | --- | --- |");
        sb.AppendLine($"| 1 | Total Requests | {data.TotalRequests} |");
        sb.AppendLine($"| 2 | Open Requests | {data.OpenRequests} |");
        sb.AppendLine($"| 3 | Close Requests | {data.ClosedRequests} |");
        sb.AppendLine($"| 4 | Common Issues | {EscapeCell(commonIssues)} |");
        sb.AppendLine($"| 5 | Overall Summary | {EscapeCell(overallSummary)} |");
        sb.Append($"| 6 | Recommended Next Actions | {EscapeCell(recommendedActions)} |");
        return sb.ToString();
    }

    private static string EscapeCell(string value) =>
        value.Replace("|", "\\|").Replace("\r", " ").Replace("\n", " ");
}
