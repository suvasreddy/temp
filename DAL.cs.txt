public interface IRepository
{
    Task<bool> AuthenticateAsync(User user);
    Task<User?> GetUserAsync(string userName);
    Task<ServiceRequest> RaiseRequestAsync(ServiceRequest request);
    Task<IEnumerable<ServiceRequest>> ViewRequestsAsync(string? userName = null);

    Task<bool> ReOpenRequestAsync(int requestId);
    Task<bool> CloseRequestAsync(int requestId);
    Task<bool> DeleteRequestAsync(int requestId);
    Task<Dictionary<string, int>> GetSummaryRequestByUserAsync(string userName);
}

using Capstone.DAL.Models;
using Microsoft.EntityFrameworkCore;

namespace Capstone.DAL.Repository;

public class Repository : IRepository
{
    private readonly HelpDeskDbContext _context;

    public Repository(HelpDeskDbContext context)
    {
        _context = context;
    }

    public async Task<bool> AuthenticateAsync(User user)
    {
        if (string.IsNullOrWhiteSpace(user.UserName) || string.IsNullOrWhiteSpace(user.Password))
        {
            return false;
        }

        return await _context.Users
            .AsNoTracking()
            .AnyAsync(u => u.UserName == user.UserName && u.Password == user.Password);
    }

    public async Task<User?> GetUserAsync(string userName)
    {
        return await _context.Users
            .AsNoTracking()
            .Include(u => u.Role)
            .FirstOrDefaultAsync(u => u.UserName == userName);
    }

    public async Task<ServiceRequest> RaiseRequestAsync(ServiceRequest request)
    {
        request.RaisedOn = DateTime.UtcNow;
        if (request.ReqStatus == 0)
        {
            request.ReqStatus = 1;
        }

        _context.ServiceRequests.Add(request);
        await _context.SaveChangesAsync();
        return request;
    }

    public async Task<IEnumerable<ServiceRequest>> ViewRequestsAsync(string? userName = null)
    {
        var query = _context.ServiceRequests
            .AsNoTracking()
            .Include(sr => sr.RaisedByUser)
            .ThenInclude(u => u!.Role)
            .Include(sr => sr.Status)
            .AsQueryable();

        if (!string.IsNullOrWhiteSpace(userName))
        {
            query = query.Where(sr => sr.RaisedBy == userName);
        }

        return await query.OrderByDescending(sr => sr.RaisedOn).ToListAsync();
    }

    public async Task<ServiceRequest?> GetRequestByIdAsync(int requestId)
    {
        return await _context.ServiceRequests
            .AsNoTracking()
            .Include(sr => sr.RaisedByUser)
            .ThenInclude(u => u!.Role)
            .Include(sr => sr.Status)
            .FirstOrDefaultAsync(sr => sr.RequestId == requestId);
    }

    public async Task<bool> ReOpenRequestAsync(int requestId)
    {
        var request = await _context.ServiceRequests.FindAsync(requestId);
        if (request is null)
        {
            return false;
        }

        request.ReqStatus = 4;
        await _context.SaveChangesAsync();
        return true;
    }

    public async Task<bool> CloseRequestAsync(int requestId)
    {
        var request = await _context.ServiceRequests.FindAsync(requestId);
        if (request is null)
        {
            return false;
        }

        request.ReqStatus = 3;
        await _context.SaveChangesAsync();
        return true;
    }

    public async Task<bool> DeleteRequestAsync(int requestId)
    {
        var request = await _context.ServiceRequests.FindAsync(requestId);
        if (request is null)
        {
            return false;
        }

        _context.ServiceRequests.Remove(request);
        await _context.SaveChangesAsync();
        return true;
    }

    public async Task<Dictionary<string, int>> GetSummaryRequestByUserAsync(string userName)
    {
        return await _context.ServiceRequests
            .AsNoTracking()
            .Where(sr => sr.RaisedBy == userName)
            .Include(sr => sr.Status)
            .GroupBy(sr => sr.Status != null ? sr.Status.Description : "Unknown")
            .ToDictionaryAsync(g => g.Key, g => g.Count());
    }
}

using Capstone.DAL.Models;
using Microsoft.EntityFrameworkCore;

namespace Capstone.DAL.Repository;

public class HelpDeskDbContext : DbContext
{
    public HelpDeskDbContext(DbContextOptions<HelpDeskDbContext> options)
        : base(options)
    {
    }

    public DbSet<Role> Roles => Set<Role>();
    public DbSet<User> Users => Set<User>();
    public DbSet<Status> Statuses => Set<Status>();
    public DbSet<ServiceRequest> ServiceRequests => Set<ServiceRequest>();

    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        base.OnModelCreating(modelBuilder);

        modelBuilder.Entity<Role>().ToTable("Role");
        modelBuilder.Entity<User>().ToTable("User");
        modelBuilder.Entity<Status>().ToTable("Status");
        modelBuilder.Entity<ServiceRequest>().ToTable("ServiceRequest");

        modelBuilder.Entity<User>()
            .HasOne(u => u.Role)
            .WithMany(r => r.Users)
            .HasForeignKey(u => u.RoleId)
            .OnDelete(DeleteBehavior.Restrict);

        modelBuilder.Entity<ServiceRequest>()
            .HasOne(sr => sr.RaisedByUser)
            .WithMany(u => u.ServiceRequests)
            .HasForeignKey(sr => sr.RaisedBy)
            .OnDelete(DeleteBehavior.Restrict);

        modelBuilder.Entity<ServiceRequest>()
            .HasOne(sr => sr.Status)
            .WithMany(s => s.ServiceRequests)
            .HasForeignKey(sr => sr.ReqStatus)
            .OnDelete(DeleteBehavior.Restrict);

        modelBuilder.Entity<Role>().HasData(
            new Role { RoleId = 1, RoleName = "Admin" },
            new Role { RoleId = 2, RoleName = "User" }
        );

        modelBuilder.Entity<Status>().HasData(
            new Status { StatusId = 1, Description = "Open" },
            new Status { StatusId = 2, Description = "In Progress" },
            new Status { StatusId = 3, Description = "Closed" },
            new Status { StatusId = 4, Description = "Re-Opened" }
        );
    }
}

using Capstone.DAL.Models;
using Capstone.DAL.Repository;
using Microsoft.Data.Sqlite;
using Microsoft.EntityFrameworkCore;
using Xunit;

namespace Capstone.DAL.Tests.Repository;

public sealed class HelpDeskDbContextTests : IDisposable
{
    private readonly SqliteConnection _connection;
    private readonly HelpDeskDbContext _context;

    public HelpDeskDbContextTests()
    {
        _connection = new SqliteConnection("DataSource=:memory:");
        _connection.Open();

        DbContextOptions<HelpDeskDbContext> options = new DbContextOptionsBuilder<HelpDeskDbContext>()
            .UseSqlite(_connection)
            .Options;

        _context = new HelpDeskDbContext(options);
        _context.Database.EnsureCreated();
    }

    [Fact]
    public async Task OnModelCreating_ShouldSeedExpectedRolesAndStatuses()
    {
        List<Role> roles = await _context.Roles
            .OrderBy(r => r.RoleId)
            .ToListAsync();

        List<Status> statuses = await _context.Statuses
            .OrderBy(s => s.StatusId)
            .ToListAsync();

        Assert.Equal(2, roles.Count);
        Assert.Equal("Admin", roles[0].RoleName);
        Assert.Equal("User", roles[1].RoleName);

        Assert.Equal(4, statuses.Count);
        Assert.Equal("Open", statuses[0].Description);
        Assert.Equal("In Progress", statuses[1].Description);
        Assert.Equal("Closed", statuses[2].Description);
        Assert.Equal("Re-Opened", statuses[3].Description);
    }

    [Fact]
    public async Task SaveChangesAsync_ShouldPersistSingularTablesAndNavigationGraph()
    {
        User user = new()
        {
            UserName = "alice",
            Password = "Password123",
            CreatedOn = DateTime.UtcNow,
            RoleId = 2
        };
        _context.Users.Add(user);
        await _context.SaveChangesAsync();

        ServiceRequest request = new()
        {
            Description = "VPN setup",
            Details = "Requesting remote VPN access",
            RaisedBy = "alice",
            RaisedOn = DateTime.UtcNow,
            Justification = "Remote sprint work",
            ReqStatus = 1
        };
        _context.ServiceRequests.Add(request);
        await _context.SaveChangesAsync();

        ServiceRequest? loaded = await _context.ServiceRequests
            .AsNoTracking()
            .Include(sr => sr.RaisedByUser)
                .ThenInclude(u => u!.Role)
            .Include(sr => sr.Status)
            .FirstOrDefaultAsync(sr => sr.RequestId == request.RequestId);

        Assert.NotNull(loaded);
        Assert.Equal("alice", loaded!.RaisedByUser?.UserName);
        Assert.Equal("User", loaded.RaisedByUser?.Role?.RoleName);
        Assert.Equal("Open", loaded.Status?.Description);
    }

    [Fact]
    public async Task SaveChangesAsync_WhenDeletingReferencedRole_ShouldThrowDbUpdateExceptionDueToRestrict()
    {
        _context.Users.Add(new User
        {
            UserName = "bob",
            Password = "Password123",
            CreatedOn = DateTime.UtcNow,
            RoleId = 2
        });
        await _context.SaveChangesAsync();

        Role role = await _context.Roles.FirstAsync(r => r.RoleId == 2);
        _context.Roles.Remove(role);

        await Assert.ThrowsAsync<DbUpdateException>(() => _context.SaveChangesAsync());
    }

    public void Dispose()
    {
        _context.Dispose();
        _connection.Dispose();
    }
}

using Capstone.DAL.Models;
using Capstone.DAL.Repository;
using Microsoft.Data.Sqlite;
using Microsoft.EntityFrameworkCore;
using Xunit;

namespace Capstone.DAL.Tests.Repository;

public sealed class RepositoryTests : IDisposable
{
    private readonly SqliteConnection _connection;
    private readonly HelpDeskDbContext _context;
    private readonly IRepository _repository;

    public RepositoryTests()
    {
        _connection = new SqliteConnection("DataSource=:memory:");
        _connection.Open();

        DbContextOptions<HelpDeskDbContext> options = new DbContextOptionsBuilder<HelpDeskDbContext>()
            .UseSqlite(_connection)
            .Options;

        _context = new HelpDeskDbContext(options);
        _context.Database.EnsureCreated();

        _context.Users.AddRange(
            new User
            {
                UserName = "alice",
                Password = "Password123",
                CreatedOn = DateTime.UtcNow,
                RoleId = 2
            },
            new User
            {
                UserName = "admin",
                Password = "AdminPassword123",
                CreatedOn = DateTime.UtcNow,
                RoleId = 1
            });
        _context.SaveChanges();

        _repository = new Capstone.DAL.Repository.Repository(_context);
    }

    [Fact]
    public async Task AuthenticateAsync_WithValidCredentials_ShouldReturnTrue()
    {
        bool result = await _repository.AuthenticateAsync(new User
        {
            UserName = "alice",
            Password = "Password123"
        });

        Assert.True(result);
    }

    [Theory]
    [InlineData("alice", "WrongPassword")]
    [InlineData("unknown_user", "Password123")]
    [InlineData("", "Password123")]
    [InlineData("alice", "")]
    public async Task AuthenticateAsync_WithInvalidOrMissingCredentials_ShouldReturnFalse(string userName, string password)
    {
        bool result = await _repository.AuthenticateAsync(new User
        {
            UserName = userName,
            Password = password
        });

        Assert.False(result);
    }

    [Fact]
    public async Task GetUserAsync_WhenUserExists_ShouldReturnUserWithRoleNavigation()
    {
        User? user = await _repository.GetUserAsync("admin");

        Assert.NotNull(user);
        Assert.Equal("admin", user!.UserName);
        Assert.Equal("Admin", user.Role?.RoleName);
    }

    [Fact]
    public async Task GetUserAsync_WhenUserDoesNotExist_ShouldReturnNull()
    {
        User? user = await _repository.GetUserAsync("does_not_exist");

        Assert.Null(user);
    }

    [Fact]
    public async Task RaiseRequestAsync_ShouldDefaultStatusToOpenAndAssignId()
    {
        ServiceRequest created = await _repository.RaiseRequestAsync(new ServiceRequest
        {
            Description = "Monitor replacement",
            Details = "Flickering display on secondary monitor",
            RaisedBy = "alice",
            Justification = "Daily development work"
        });

        Assert.True(created.RequestId > 0);
        Assert.Equal(1, created.ReqStatus);
        Assert.NotEqual(default, created.RaisedOn);
    }

    [Fact]
    public async Task ViewRequestsAsync_ShouldFilterByUserNameWhenProvidedOrReturnAll()
    {
        await _repository.RaiseRequestAsync(new ServiceRequest
        {
            Description = "Alice request",
            Details = "Details for Alice",
            RaisedBy = "alice",
            Justification = "Work"
        });

        await _repository.RaiseRequestAsync(new ServiceRequest
        {
            Description = "Admin request",
            Details = "Details for Admin",
            RaisedBy = "admin",
            Justification = "Operations"
        });

        List<ServiceRequest> allRequests = (await _repository.ViewRequestsAsync()).ToList();
        List<ServiceRequest> aliceRequests = (await _repository.ViewRequestsAsync("alice")).ToList();

        Assert.Equal(2, allRequests.Count);
        Assert.Single(aliceRequests);
        Assert.Equal("alice", aliceRequests[0].RaisedBy);
    }

    [Fact]
    public async Task CloseAndReOpenRequestAsync_ShouldUpdateStatusCodes()
    {
        ServiceRequest created = await _repository.RaiseRequestAsync(new ServiceRequest
        {
            Description = "Access issue",
            Details = "Repo permission needed",
            RaisedBy = "alice",
            Justification = "Feature branch"
        });

        bool closed = await _repository.CloseRequestAsync(created.RequestId);
        ServiceRequest? afterClose = await _repository.GetRequestByIdAsync(created.RequestId);

        bool reopened = await _repository.ReOpenRequestAsync(created.RequestId);
        ServiceRequest? afterReopen = await _repository.GetRequestByIdAsync(created.RequestId);

        Assert.True(closed);
        Assert.Equal(3, afterClose?.ReqStatus);
        Assert.True(reopened);
        Assert.Equal(4, afterReopen?.ReqStatus);
    }

    [Fact]
    public async Task GetSummaryRequestByUserAsync_ShouldGroupCountsByStatusDescription()
    {
        ServiceRequest first = await _repository.RaiseRequestAsync(new ServiceRequest
        {
            Description = "Request 1",
            Details = "First request",
            RaisedBy = "alice",
            Justification = "Task 1"
        });

        ServiceRequest second = await _repository.RaiseRequestAsync(new ServiceRequest
        {
            Description = "Request 2",
            Details = "Second request",
            RaisedBy = "alice",
            Justification = "Task 2"
        });

        await _repository.CloseRequestAsync(first.RequestId);
        await _repository.ReOpenRequestAsync(second.RequestId);

        Dictionary<string, int> summary = await _repository.GetSummaryRequestByUserAsync("alice");

        Assert.Equal(1, summary["Closed"]);
        Assert.Equal(1, summary["Re-Opened"]);
    }

    [Fact]
    public async Task DeleteRequestAsync_WhenRequestExists_ShouldRemoveRecord()
    {
        ServiceRequest created = await _repository.RaiseRequestAsync(new ServiceRequest
        {
            Description = "Temporary ticket",
            Details = "To be deleted",
            RaisedBy = "alice",
            Justification = "Cleanup test"
        });

        bool deleted = await _repository.DeleteRequestAsync(created.RequestId);
        ServiceRequest? loaded = await _repository.GetRequestByIdAsync(created.RequestId);

        Assert.True(deleted);
        Assert.Null(loaded);
    }

    [Fact]
    public async Task MissingRequestId_ShouldReturnFalseForStateChangesAndDelete()
    {
        const int nonExistentId = 99999;

        Assert.False(await _repository.CloseRequestAsync(nonExistentId));
        Assert.False(await _repository.ReOpenRequestAsync(nonExistentId));
        Assert.False(await _repository.DeleteRequestAsync(nonExistentId));
        Assert.Null(await _repository.GetRequestByIdAsync(nonExistentId));
    }

    public void Dispose()
    {
        _context.Dispose();
        _connection.Dispose();
    }
}

[Fact]
public async Task SaveChangesAsync_WhenDeletingReferencedRole_ShouldThrowDbUpdateExceptionDueToRestrict()
{
    User user = new()
    {
        UserName = "bob",
        Password = "Password123",
        CreatedOn = DateTime.UtcNow,
        RoleId = 2
    };
    _context.Users.Add(user);
    await _context.SaveChangesAsync();

    // Detach tracked user so EF Core sends DELETE to SQLite on SaveChangesAsync
    _context.Entry(user).State = EntityState.Detached;

    Role role = await _context.Roles.FirstAsync(r => r.RoleId == 2);
    _context.Roles.Remove(role);

    await Assert.ThrowsAsync<DbUpdateException>(() => _context.SaveChangesAsync());
}
