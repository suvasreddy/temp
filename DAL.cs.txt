using System.Collections.Generic;
using System.Threading.Tasks;
using Capstone.API.DTOs.ServiceRequest;
using Capstone.API.DTOs.Summary;

namespace Capstone.API.Services.Interfaces;

public interface IRequestService
{
    IEnumerable<RequestListItemDto> GetRequests(string actingUserName);

    ServiceRequestResponseDto? GetRequestById(int requestId, string actingUserName);

    ServiceRequestResponseDto? CreateRequest(CreateServiceRequestDto dto, string actingUserName);

    bool DeleteRequest(int requestId, string actingUserName);

    ServiceRequestResponseDto? ReopenRequest(int requestId, string actingUserName);

    ServiceRequestResponseDto? CloseRequest(int requestId, string actingUserName);

    Task<SummaryDataDto?> GetUserSummaryAsync(string actingUserName, string? targetUserName = null);
}

// File: Services/Implementations/RequestService.cs

using System;
using System.Collections.Generic;
using System.Linq;
using System.Threading.Tasks;
using Capstone.API.DTOs.ServiceRequest;
using Capstone.API.DTOs.Summary;
using Capstone.API.Services.Interfaces;
using Capstone.DAL.Models;
using Capstone.DAL.Repository;

namespace Capstone.API.Services.Implementations;

public class RequestService : IRequestService
{
    private readonly IRepository _repository;
    private readonly IAuthenticationService _authService;

    public RequestService(IRepository repository, IAuthenticationService authService)
    {
        _repository = repository;
        _authService = authService;
    }

    public IEnumerable<RequestListItemDto> GetRequests(string actingUserName)
    {
        if (string.IsNullOrWhiteSpace(actingUserName) || _authService.GetUserProfile(actingUserName) is null)
        {
            return Enumerable.Empty<RequestListItemDto>();
        }

        var requests = _authService.IsAdmin(actingUserName)
            ? _repository.ViewRequests()
            : _repository.ViewRequests(actingUserName);

        return requests.Select(ToListItemDto).ToList();
    }

    public ServiceRequestResponseDto? GetRequestById(int requestId, string actingUserName)
    {
        if (requestId <= 0 || string.IsNullOrWhiteSpace(actingUserName))
        {
            return null;
        }

        var request = _repository.GetRequestById(requestId);
        if (request is null || !CanAccessRequest(request, actingUserName))
        {
            return null;
        }

        return ToResponseDto(request);
    }

    public ServiceRequestResponseDto? CreateRequest(CreateServiceRequestDto dto, string actingUserName)
    {
        if (dto is null || string.IsNullOrWhiteSpace(actingUserName) || _authService.GetUserProfile(actingUserName) is null)
        {
            return null;
        }

        var entity = new ServiceRequest
        {
            Description = dto.Description,
            Details = dto.Details,
            Justification = dto.Justification,
            RaisedBy = actingUserName,
            RaisedOn = DateTime.UtcNow,
            ReqStatus = 1
        };

        var createdId = _repository.RaiseRequest(entity);
        entity.RequestId = createdId;

        return _repository.GetRequestById(createdId) is { } saved ? ToResponseDto(saved) : ToResponseDto(entity);
    }

    public bool DeleteRequest(int requestId, string actingUserName)
    {
        if (requestId <= 0 || string.IsNullOrWhiteSpace(actingUserName))
        {
            return false;
        }

        var request = _repository.GetRequestById(requestId);
        if (request is null || !IsOwner(request, actingUserName))
        {
            return false;
        }

        return _repository.DeleteRequest(requestId);
    }

    public ServiceRequestResponseDto? ReopenRequest(int requestId, string actingUserName)
    {
        if (requestId <= 0 || string.IsNullOrWhiteSpace(actingUserName))
        {
            return null;
        }

        var request = _repository.GetRequestById(requestId);
        if (request is null || !IsOwner(request, actingUserName))
        {
            return null;
        }

        if (!_repository.ReOpenRequest(request))
        {
            return null;
        }

        return _repository.GetRequestById(requestId) is { } updated ? ToResponseDto(updated) : ToResponseDto(new ServiceRequest
        {
            RequestId = request.RequestId,
            Description = request.Description,
            Details = request.Details,
            RaisedBy = request.RaisedBy,
            RaisedOn = request.RaisedOn,
            Justification = request.Justification,
            ReqStatus = 4,
            Status = request.Status
        });
    }

    public ServiceRequestResponseDto? CloseRequest(int requestId, string actingUserName)
    {
        if (requestId <= 0 || string.IsNullOrWhiteSpace(actingUserName) || !_authService.IsAdmin(actingUserName))
        {
            return null;
        }

        var request = _repository.GetRequestById(requestId);
        if (request is null)
        {
            return null;
        }

        if (!_repository.CloseRequest(requestId))
        {
            return null;
        }

        return _repository.GetRequestById(requestId) is { } updated ? ToResponseDto(updated) : ToResponseDto(new ServiceRequest
        {
            RequestId = request.RequestId,
            Description = request.Description,
            Details = request.Details,
            RaisedBy = request.RaisedBy,
            RaisedOn = request.RaisedOn,
            Justification = request.Justification,
            ReqStatus = 3,
            Status = request.Status
        });
    }

    public async Task<SummaryDataDto?> GetUserSummaryAsync(string actingUserName, string? targetUserName = null)
    {
        if (string.IsNullOrWhiteSpace(actingUserName) || !_authService.IsAdmin(actingUserName))
        {
            return null;
        }

        var effectiveUser = string.IsNullOrWhiteSpace(targetUserName) ? actingUserName : targetUserName;
        var summary = await _repository.GetSummaryRequestByUserAsync(effectiveUser);

        return new SummaryDataDto
        {
            UserName = summary.UserName,
            TotalRequests = summary.TotalRequests,
            OpenRequests = summary.OpenRequests,
            InProgressRequests = summary.InProgressRequests,
            ClosedRequests = summary.ClosedRequests,
            ReOpenedRequests = summary.ReOpenedRequests,
            Requests = summary.Requests.Select(r => new RequestSummaryItemDto
            {
                RequestId = r.RequestId,
                Description = r.Description,
                Status = r.Status,
                RaisedOn = r.RaisedOn
            }).ToList()
        };
    }

    private bool CanAccessRequest(ServiceRequest request, string actingUserName)
    {
        return _authService.IsAdmin(actingUserName) || IsOwner(request, actingUserName);
    }

    private static bool IsOwner(ServiceRequest request, string actingUserName)
    {
        return string.Equals(request.RaisedBy, actingUserName, StringComparison.OrdinalIgnoreCase);
    }

    private static RequestListItemDto ToListItemDto(ServiceRequest request)
    {
        return new RequestListItemDto
        {
            RequestId = request.RequestId,
            Description = request.Description,
            RaisedBy = request.RaisedBy,
            RaisedOn = request.RaisedOn,
            StatusDescription = request.Status?.Description ?? GetStatusFallback(request.ReqStatus)
        };
    }

    private static ServiceRequestResponseDto ToResponseDto(ServiceRequest request)
    {
        return new ServiceRequestResponseDto
        {
            RequestId = request.RequestId,
            Description = request.Description,
            Details = request.Details,
            RaisedBy = request.RaisedBy,
            RaisedOn = request.RaisedOn,
            Justification = request.Justification,
            ReqStatus = request.ReqStatus,
            StatusDescription = request.Status?.Description ?? GetStatusFallback(request.ReqStatus)
        };
    }

    private static string GetStatusFallback(int statusId)
    {
        return statusId switch
        {
            1 => "Open",
            2 => "In Progress",
            3 => "Closed",
            4 => "Reopened",
            _ => "Unknown"
        };
    }
}

    [Fact]
    public void GetRequests_Admin_CallsViewRequestsWithoutFilter()
    {
        _authServiceMock.Setup(a => a.GetUserProfile("admin1")).Returns(new UserResponseDto { UserName = "admin1", RoleName = "Admin" });
        _authServiceMock.Setup(a => a.IsAdmin("admin1")).Returns(true);
        _repositoryMock.Setup(r => r.ViewRequests()).Returns(new List<ServiceRequest>
        {
            new() { RequestId = 1, Description = "VPN", RaisedBy = "alice", RaisedOn = DateTime.UtcNow, ReqStatus = 1, Status = new Status { StatusId = 1, Description = "Open" } },
            new() { RequestId = 2, Description = "Mouse", RaisedBy = "bob", RaisedOn = DateTime.UtcNow, ReqStatus = 2 }
        });

        var result = _sut.GetRequests("admin1").ToList();

        Assert.Equal(2, result.Count);
        Assert.Equal("Open", result[0].StatusDescription);
        Assert.Equal("In Progress", result[1].StatusDescription);
