using Capstone.API.DTOs.Authentication;
using Capstone.API.DTOs.User;

namespace Capstone.API.Services.Interfaces;

public interface IAuthenticationService
{
    LoginResponseDto? Login(LoginRequestDto request);

    UserResponseDto? GetUserProfile(string userName);

    bool IsAdmin(string userName);
}

// =============================================================
// File: Services/Implementations/AuthenticationService.cs
// =============================================================
using System;
using Capstone.API.DTOs.Authentication;
using Capstone.API.DTOs.User;
using Capstone.API.Services.Interfaces;
using Capstone.DAL.Repository;

namespace Capstone.API.Services.Implementations;

public class AuthenticationService : IAuthenticationService
{
    private readonly IRepository _repository;

    public AuthenticationService(IRepository repository)
    {
        _repository = repository;
    }

    public LoginResponseDto? Login(LoginRequestDto request)
    {
        if (string.IsNullOrWhiteSpace(request.UserName) || string.IsNullOrWhiteSpace(request.Password))
        {
            return null;
        }

        var user = _repository.GetUser(request.UserName);
        if (user is null || user.Password != request.Password)
        {
            return null;
        }

        return new LoginResponseDto
        {
            Token = string.Empty,
            UserName = user.UserName,
            Role = user.Role?.RoleName ?? string.Empty,
            ExpiresAt = DateTime.UtcNow.AddHours(2)
        };
    }

    public UserResponseDto? GetUserProfile(string userName)
    {
        if (string.IsNullOrWhiteSpace(userName))
        {
            return null;
        }

        var user = _repository.GetUser(userName);
        if (user is null)
        {
            return null;
        }

        return new UserResponseDto
        {
            UserName = user.UserName,
            RoleId = user.RoleId,
            RoleName = user.Role?.RoleName ?? string.Empty,
            CreatedOn = user.CreatedOn
        };
    }

    public bool IsAdmin(string userName)
    {
        if (string.IsNullOrWhiteSpace(userName))
        {
            return false;
        }

        var user = _repository.GetUser(userName);
        return string.Equals(user?.Role?.RoleName, "Admin", StringComparison.OrdinalIgnoreCase);
    }
}

using System;
using Capstone.API.DTOs.Authentication;
using Capstone.API.Services.Implementations;
using Capstone.DAL.Models;
using Capstone.DAL.Repository;
using Moq;
using Xunit;

namespace Capstone.API.Tests.Services;

public class AuthenticationServiceTests
{
    private readonly Mock<IRepository> _repositoryMock = new();
    private readonly AuthenticationService _sut;

    public AuthenticationServiceTests()
    {
        _sut = new AuthenticationService(_repositoryMock.Object);
    }

    [Fact]
    public void Login_ValidCredentials_ReturnsLoginResponse()
    {
        var beforeCall = DateTime.UtcNow;
        _repositoryMock.Setup(r => r.GetUser("alice")).Returns(new User
        {
            UserName = "alice",
            Password = "Password123",
            RoleId = 2,
            Role = new Role { RoleId = 2, RoleName = "User" }
        });

        var result = _sut.Login(new LoginRequestDto
        {
            UserName = "alice",
            Password = "Password123"
        });

        Assert.NotNull(result);
        Assert.Equal(string.Empty, result!.Token);
        Assert.Equal("alice", result.UserName);
        Assert.Equal("User", result.Role);
        Assert.True(result.ExpiresAt > beforeCall);
    }

    [Theory]
    [InlineData("", "Password123")]
    [InlineData("alice", "")]
    [InlineData("alice", "WrongPass")]
    public void Login_InvalidInputOrPassword_ReturnsNull(string userName, string password)
    {
        _repositoryMock.Setup(r => r.GetUser("alice")).Returns(new User
        {
            UserName = "alice",
            Password = "Password123",
            Role = new Role { RoleId = 2, RoleName = "User" }
        });

        var result = _sut.Login(new LoginRequestDto
        {
            UserName = userName,
            Password = password
        });

        Assert.Null(result);
    }

    [Fact]
    public void GetUserProfile_ExistingUser_ReturnsProfileWithoutPassword()
    {
        var createdOn = new DateTime(2026, 1, 10, 10, 0, 0, DateTimeKind.Utc);
        _repositoryMock.Setup(r => r.GetUser("admin1")).Returns(new User
        {
            UserName = "admin1",
            Password = "AdminPassword123",
            RoleId = 1,
            Role = new Role { RoleId = 1, RoleName = "Admin" },
            CreatedOn = createdOn
        });

        var profile = _sut.GetUserProfile("admin1");

        Assert.NotNull(profile);
        Assert.Equal("admin1", profile!.UserName);
        Assert.Equal(1, profile.RoleId);
        Assert.Equal("Admin", profile.RoleName);
        Assert.Equal(createdOn, profile.CreatedOn);
    }

    [Fact]
    public void IsAdmin_AdminRole_ReturnsTrueAndNonAdmin_ReturnsFalse()
    {
        _repositoryMock.Setup(r => r.GetUser("admin1")).Returns(new User
        {
            UserName = "admin1",
            Role = new Role { RoleId = 1, RoleName = "Admin" }
        });
        _repositoryMock.Setup(r => r.GetUser("user1")).Returns(new User
        {
            UserName = "user1",
            Role = new Role { RoleId = 2, RoleName = "User" }
        });

        Assert.True(_sut.IsAdmin("admin1"));
        Assert.False(_sut.IsAdmin("user1"));
        Assert.False(_sut.IsAdmin("missing"));
    }
}
