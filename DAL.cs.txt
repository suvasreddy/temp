using Capstone.DAL.Models;
using Microsoft.EntityFrameworkCore;

namespace Capstone.DAL.Repository;

public class Repository : IRepository
{
    private readonly HelpDeskDbContext _context;

    public Repository(HelpDeskDbContext context)
    {
        _context = context;
    }

    public bool Authenticate(User user)
    {
        if (string.IsNullOrWhiteSpace(user.UserName) || string.IsNullOrWhiteSpace(user.Password))
        {
            return false;
        }

        return _context.Users
            .AsNoTracking()
            .Any(u => u.UserName == user.UserName && u.Password == user.Password);
    }

    public List<ServiceRequest> ViewRequests()
    {
        return _context.ServiceRequests
            .AsNoTracking()
            .Include(sr => sr.RaisedByUser)
                .ThenInclude(u => u!.Role)
            .Include(sr => sr.Status)
            .OrderByDescending(sr => sr.RaisedOn)
            .ToList();
    }

    public List<ServiceRequest> ViewRequests(string userName)
    {
        return _context.ServiceRequests
            .AsNoTracking()
            .Include(sr => sr.RaisedByUser)
                .ThenInclude(u => u!.Role)
            .Include(sr => sr.Status)
            .Where(sr => sr.RaisedBy == userName)
            .OrderByDescending(sr => sr.RaisedOn)
            .ToList();
    }

    public int RaiseRequest(ServiceRequest newRequest)
    {
        newRequest.RaisedOn = DateTime.UtcNow;
        if (newRequest.ReqStatus == 0)
        {
            newRequest.ReqStatus = 1;
        }

        _context.ServiceRequests.Add(newRequest);
        _context.SaveChanges();
        return newRequest.RequestId;
    }

    public ServiceRequest? GetRequestById(int requestId)
    {
        return _context.ServiceRequests
            .AsNoTracking()
            .Include(sr => sr.RaisedByUser)
                .ThenInclude(u => u!.Role)
            .Include(sr => sr.Status)
            .FirstOrDefault(sr => sr.RequestId == requestId);
    }

    public bool ReOpenRequest(ServiceRequest request)
    {
        var existing = _context.ServiceRequests.Find(request.RequestId);
        if (existing is null)
        {
            return false;
        }

        existing.ReqStatus = 4;
        _context.SaveChanges();
        return true;
    }

    public List<ServiceRequest> GetRequestBySP(string userName)
    {
        return _context.ServiceRequests
            .FromSqlInterpolated($"SELECT * FROM ServiceRequest WHERE RaisedBy = {userName}")
            .AsNoTracking()
            .ToList();
    }

    public bool CloseRequest(int requestId)
    {
        var request = _context.ServiceRequests.Find(requestId);
        if (request is null)
        {
            return false;
        }

        request.ReqStatus = 3;
        _context.SaveChanges();
        return true;
    }

    public bool DeleteRequest(int requestId)
    {
        var request = _context.ServiceRequests.Find(requestId);
        if (request is null)
        {
            return false;
        }

        _context.ServiceRequests.Remove(request);
        _context.SaveChanges();
        return true;
    }

    public User? GetUser(string userName)
    {
        return _context.Users
            .AsNoTracking()
            .Include(u => u.Role)
            .FirstOrDefault(u => u.UserName == userName);
    }

    public async Task<Dictionary<string, int>> GetSummaryRequestByUserAsync(string userName)
    {
        return await _context.ServiceRequests
            .AsNoTracking()
            .Where(sr => sr.RaisedBy == userName)
            .Include(sr => sr.Status)
            .GroupBy(sr => sr.Status != null ? sr.Status.Description : "Unknown")
            .ToDictionaryAsync(g => g.Key, g => g.Count());
    }
}

using Capstone.DAL.Models;
using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Design;

namespace Capstone.DAL.Repository;

public class HelpDeskDbContext : DbContext
{
    public HelpDeskDbContext(DbContextOptions<HelpDeskDbContext> options)
        : base(options)
    {
    }

    public class HelpDeskDbContextFactory : IDesignTimeDbContextFactory<HelpDeskDbContext>
    {
        public HelpDeskDbContext CreateDbContext(string[] args)
        {
            var options = new DbContextOptionsBuilder<HelpDeskDbContext>()
                .UseSqlite(@"Data Source=Database\ServiceDeskDB.db")
                .Options;

            return new HelpDeskDbContext(options);
        }
    }

    public DbSet<Role> Roles { get; set; }
    public DbSet<User> Users { get; set; }
    public DbSet<ServiceRequest> ServiceRequests { get; set; }
    public DbSet<Status> Statuses { get; set; }

    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        base.OnModelCreating(modelBuilder);

        modelBuilder.Entity<Role>().ToTable("Role");
        modelBuilder.Entity<User>().ToTable("User");
        modelBuilder.Entity<Status>().ToTable("Status");
        modelBuilder.Entity<ServiceRequest>().ToTable("ServiceRequest");

        modelBuilder.Entity<User>()
            .HasOne(u => u.Role)
            .WithMany()
            .HasForeignKey(u => u.RoleId)
            .OnDelete(DeleteBehavior.Restrict);

        modelBuilder.Entity<ServiceRequest>()
            .HasOne(sr => sr.RaisedByUser)
            .WithMany()
            .HasForeignKey(sr => sr.RaisedBy)
            .OnDelete(DeleteBehavior.Restrict);

        modelBuilder.Entity<ServiceRequest>()
            .HasOne(sr => sr.Status)
            .WithMany()
            .HasForeignKey(sr => sr.ReqStatus)
            .OnDelete(DeleteBehavior.Restrict);

        modelBuilder.Entity<Role>().HasData(
            new Role { RoleId = 1, RoleName = "Admin" },
            new Role { RoleId = 2, RoleName = "User" }
        );

        modelBuilder.Entity<Status>().HasData(
            new Status { StatusId = 1, Description = "Open" },
            new Status { StatusId = 2, Description = "In Progress" },
            new Status { StatusId = 3, Description = "Closed" },
            new Status { StatusId = 4, Description = "Re-Opened" }
        );
    }
}

using Capstone.DAL.Models;
using Capstone.DAL.Repository;
using Microsoft.Data.Sqlite;
using Microsoft.EntityFrameworkCore;
using Xunit;

namespace Capstone.DAL.Tests.Repository;

public sealed class HelpDeskDbContextTests : IDisposable
{
    private readonly SqliteConnection _connection;
    private readonly HelpDeskDbContext _context;

    public HelpDeskDbContextTests()
    {
        _connection = new SqliteConnection("DataSource=:memory:");
        _connection.Open();

        DbContextOptions<HelpDeskDbContext> options = new DbContextOptionsBuilder<HelpDeskDbContext>()
            .UseSqlite(_connection)
            .Options;

        _context = new HelpDeskDbContext(options);
        _context.Database.EnsureCreated();
    }

    [Fact]
    public async Task OnModelCreating_ShouldSeedExpectedRolesAndStatuses()
    {
        List<Role> roles = await _context.Roles
            .OrderBy(r => r.RoleId)
            .ToListAsync();

        List<Status> statuses = await _context.Statuses
            .OrderBy(s => s.StatusId)
            .ToListAsync();

        Assert.Equal(2, roles.Count);
        Assert.Equal("Admin", roles[0].RoleName);
        Assert.Equal("User", roles[1].RoleName);

        Assert.Equal(4, statuses.Count);
        Assert.Equal("Open", statuses[0].Description);
        Assert.Equal("In Progress", statuses[1].Description);
        Assert.Equal("Closed", statuses[2].Description);
        Assert.Equal("Re-Opened", statuses[3].Description);
    }

    [Fact]
    public async Task SaveChangesAsync_ShouldPersistSingularTablesAndNavigationGraph()
    {
        User user = new()
        {
            UserName = "alice",
            Password = "Password123",
            CreatedOn = DateTime.UtcNow,
            RoleId = 2
        };
        _context.Users.Add(user);
        await _context.SaveChangesAsync();

        ServiceRequest request = new()
        {
            Description = "VPN setup",
            Details = "Requesting remote VPN access",
            RaisedBy = "alice",
            RaisedOn = DateTime.UtcNow,
            Justification = "Remote sprint work",
            ReqStatus = 1
        };
        _context.ServiceRequests.Add(request);
        await _context.SaveChangesAsync();

        ServiceRequest? loaded = await _context.ServiceRequests
            .AsNoTracking()
            .Include(sr => sr.RaisedByUser)
                .ThenInclude(u => u!.Role)
            .Include(sr => sr.Status)
            .FirstOrDefaultAsync(sr => sr.RequestId == request.RequestId);

        Assert.NotNull(loaded);
        Assert.Equal("alice", loaded!.RaisedByUser?.UserName);
        Assert.Equal("User", loaded.RaisedByUser?.Role?.RoleName);
        Assert.Equal("Open", loaded.Status?.Description);
    }

    [Fact]
    public async Task SaveChangesAsync_WhenDeletingReferencedRole_ShouldThrowDbUpdateExceptionDueToRestrict()
    {
        User user = new()
        {
            UserName = "bob",
            Password = "Password123",
            CreatedOn = DateTime.UtcNow,
            RoleId = 2
        };
        _context.Users.Add(user);
        await _context.SaveChangesAsync();

        _context.Entry(user).State = EntityState.Detached;

        Role role = await _context.Roles.FirstAsync(r => r.RoleId == 2);
        _context.Roles.Remove(role);

        await Assert.ThrowsAsync<DbUpdateException>(() => _context.SaveChangesAsync());
    }

    public void Dispose()
    {
        _context.Dispose();
        _connection.Dispose();
    }
}

using Capstone.DAL.Models;
using Capstone.DAL.Repository;
using Microsoft.Data.Sqlite;
using Microsoft.EntityFrameworkCore;
using Xunit;

namespace Capstone.DAL.Tests.Repository;

public sealed class RepositoryTests : IDisposable
{
    private readonly SqliteConnection _connection;
    private readonly HelpDeskDbContext _context;
    private readonly IRepository _repository;

    public RepositoryTests()
    {
        _connection = new SqliteConnection("DataSource=:memory:");
        _connection.Open();

        DbContextOptions<HelpDeskDbContext> options = new DbContextOptionsBuilder<HelpDeskDbContext>()
            .UseSqlite(_connection)
            .Options;

        _context = new HelpDeskDbContext(options);
        _context.Database.EnsureCreated();

        _context.Users.AddRange(
            new User
            {
                UserName = "alice",
                Password = "Password123",
                CreatedOn = DateTime.UtcNow,
                RoleId = 2
            },
            new User
            {
                UserName = "admin",
                Password = "AdminPassword123",
                CreatedOn = DateTime.UtcNow,
                RoleId = 1
            });
        _context.SaveChanges();

        _repository = new Capstone.DAL.Repository.Repository(_context);
    }

    [Fact]
    public void Authenticate_WithValidCredentials_ShouldReturnTrue()
    {
        bool result = _repository.Authenticate(new User
        {
            UserName = "alice",
            Password = "Password123"
        });

        Assert.True(result);
    }

    [Theory]
    [InlineData("alice", "WrongPassword")]
    [InlineData("unknown_user", "Password123")]
    [InlineData("", "Password123")]
    [InlineData("alice", "")]
    public void Authenticate_WithInvalidOrMissingCredentials_ShouldReturnFalse(string userName, string password)
    {
        bool result = _repository.Authenticate(new User
        {
            UserName = userName,
            Password = password
        });

        Assert.False(result);
    }

    [Fact]
    public void GetUser_WhenUserExists_ShouldReturnUserWithRoleNavigation()
    {
        User? user = _repository.GetUser("admin");

        Assert.NotNull(user);
        Assert.Equal("admin", user!.UserName);
        Assert.Equal("Admin", user.Role?.RoleName);
    }

    [Fact]
    public void GetUser_WhenUserDoesNotExist_ShouldReturnNull()
    {
        User? user = _repository.GetUser("does_not_exist");

        Assert.Null(user);
    }

    [Fact]
    public void RaiseRequest_ShouldDefaultStatusToOpenAndReturnNewId()
    {
        ServiceRequest request = new()
        {
            Description = "Monitor replacement",
            Details = "Flickering display on secondary monitor",
            RaisedBy = "alice",
            Justification = "Daily development work"
        };

        int createdId = _repository.RaiseRequest(request);
        ServiceRequest? loaded = _repository.GetRequestById(createdId);

        Assert.True(createdId > 0);
        Assert.NotNull(loaded);
        Assert.Equal(1, loaded!.ReqStatus);
        Assert.NotEqual(default, loaded.RaisedOn);
    }

    [Fact]
    public void ViewRequests_ShouldReturnAllOrFilterByUserName()
    {
        _repository.RaiseRequest(new ServiceRequest
        {
            Description = "Alice request",
            Details = "Details for Alice",
            RaisedBy = "alice",
            Justification = "Work"
        });

        _repository.RaiseRequest(new ServiceRequest
        {
            Description = "Admin request",
            Details = "Details for Admin",
            RaisedBy = "admin",
            Justification = "Operations"
        });

        List<ServiceRequest> allRequests = _repository.ViewRequests();
        List<ServiceRequest> aliceRequests = _repository.ViewRequests("alice");

        Assert.Equal(2, allRequests.Count);
        Assert.Single(aliceRequests);
        Assert.Equal("alice", aliceRequests[0].RaisedBy);
    }

    [Fact]
    public void GetRequestBySP_ShouldReturnRequestsForUser()
    {
        _repository.RaiseRequest(new ServiceRequest
        {
            Description = "Alice ticket 1",
            Details = "Details 1",
            RaisedBy = "alice",
            Justification = "Support"
        });

        _repository.RaiseRequest(new ServiceRequest
        {
            Description = "Admin ticket",
            Details = "Details 2",
            RaisedBy = "admin",
            Justification = "Ops"
        });

        List<ServiceRequest> aliceRequests = _repository.GetRequestBySP("alice");

        Assert.Single(aliceRequests);
        Assert.Equal("alice", aliceRequests[0].RaisedBy);
    }

    [Fact]
    public void CloseAndReOpenRequest_ShouldUpdateStatusCodes()
    {
        int createdId = _repository.RaiseRequest(new ServiceRequest
        {
            Description = "Access issue",
            Details = "Repo permission needed",
            RaisedBy = "alice",
            Justification = "Feature branch"
        });

        bool closed = _repository.CloseRequest(createdId);
        ServiceRequest? afterClose = _repository.GetRequestById(createdId);

        bool reopened = _repository.ReOpenRequest(new ServiceRequest { RequestId = createdId });
        ServiceRequest? afterReopen = _repository.GetRequestById(createdId);

        Assert.True(closed);
        Assert.Equal(3, afterClose?.ReqStatus);
        Assert.True(reopened);
        Assert.Equal(4, afterReopen?.ReqStatus);
    }

    [Fact]
    public async Task GetSummaryRequestByUserAsync_ShouldGroupCountsByStatusDescription()
    {
        int firstId = _repository.RaiseRequest(new ServiceRequest
        {
            Description = "Request 1",
            Details = "First request",
            RaisedBy = "alice",
            Justification = "Task 1"
        });

        int secondId = _repository.RaiseRequest(new ServiceRequest
        {
            Description = "Request 2",
            Details = "Second request",
            RaisedBy = "alice",
            Justification = "Task 2"
        });

        _repository.CloseRequest(firstId);
        _repository.ReOpenRequest(new ServiceRequest { RequestId = secondId });

        Dictionary<string, int> summary = await _repository.GetSummaryRequestByUserAsync("alice");

        Assert.Equal(1, summary["Closed"]);
        Assert.Equal(1, summary["Re-Opened"]);
    }

    [Fact]
    public void DeleteRequest_WhenRequestExists_ShouldRemoveRecord()
    {
        int createdId = _repository.RaiseRequest(new ServiceRequest
        {
            Description = "Temporary ticket",
            Details = "To be deleted",
            RaisedBy = "alice",
            Justification = "Cleanup test"
        });

        bool deleted = _repository.DeleteRequest(createdId);
        ServiceRequest? loaded = _repository.GetRequestById(createdId);

        Assert.True(deleted);
        Assert.Null(loaded);
    }

    [Fact]
    public void MissingRequestId_ShouldReturnFalseOrNullForOperations()
    {
        const int nonExistentId = 99999;

        Assert.False(_repository.CloseRequest(nonExistentId));
        Assert.False(_repository.ReOpenRequest(new ServiceRequest { RequestId = nonExistentId }));
        Assert.False(_repository.DeleteRequest(nonExistentId));
        Assert.Null(_repository.GetRequestById(nonExistentId));
    }

    public void Dispose()
    {
        _context.Dispose();
        _connection.Dispose();
    }
}
